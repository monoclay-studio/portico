# ── Reviewer probes (zz-review; scratch copy only). Each run is EXPECTED TO PASS = the guard does NOT fire. ──
# compute/main.tf:46 maps each injected ENV name to integration_secret_arns[key], so ENV=TEST name + key=-live means
# the container's *_TEST variable is filled from the LIVE secret.

run "zz_review_live_secret_under_test_name_in_staging" {
  command = plan
  variables {
    tier            = "lean"
    deployment      = "staging"
    domain_name     = "portico.example"
    route53_zone_id = "Z0000000000000000000"
    injected_integration_secrets = {
      PAYMENTS_STRIPE_SECRET_KEY_TEST      = "payments-stripe-connect-secret-key-live"
      PAYMENTS_STRIPE_WEBHOOK_SECRETS_TEST = "payments-stripe-connect-webhook-secrets-live"
    }
  }
  assert {
    condition     = contains(output.runtime_contract.api.secrets, "PAYMENTS_STRIPE_SECRET_KEY_TEST") && contains(output.runtime_contract.worker.secrets, "PAYMENTS_STRIPE_WEBHOOK_SECRETS_TEST")
    error_message = "plan accepted: staging tasks get the LIVE Connect key and webhook secrets under the TEST names."
  }
}

# Staging (and every managed deployment) runs NODE_ENV=production; apps/api/src/composition.ts:248/271 then disables
# the operator console, so the operator placeholders created and injectable outside production can never work.
run "zz_review_staging_operator_secrets_accepted_but_node_env_production" {
  command = plan
  variables {
    tier            = "lean"
    deployment      = "staging"
    domain_name     = "portico.example"
    route53_zone_id = "Z0000000000000000000"
    injected_integration_secrets = { OPERATOR_TOKEN = "operator-token", PORTICO_OPERATORS = "portico-operators" }
  }
  assert {
    condition     = contains(output.runtime_contract.api.environment, "NODE_ENV") && contains(output.runtime_contract.api.secrets, "PORTICO_OPERATORS")
    error_message = "staging api: NODE_ENV (=production, compute/main.tf:26) together with the operator secrets."
  }
}
