function off(instant, timeZone){const p=Object.fromEntries(new Intl.DateTimeFormat('en-US',{timeZone,hourCycle:'h23',year:'numeric',month:'2-digit',day:'2-digit',hour:'2-digit',minute:'2-digit',second:'2-digit'}).formatToParts(instant).map(x=>[x.type,x.value]));const a=Date.UTC(+p.year,+p.month-1,+p.day,+p.hour,+p.minute,+p.second);return Math.round((a-Math.floor(instant.getTime()/1000)*1000)/60000)}
function z2i(date,t,tz){const n=Date.parse(`${date}T${t}:00Z`);let g=n-off(new Date(n),tz)*60000;g=n-off(new Date(g),tz)*60000;return new Date(g)}
function i2z(i,tz){const p=Object.fromEntries(new Intl.DateTimeFormat('en-CA',{timeZone:tz,hourCycle:'h23',year:'numeric',month:'2-digit',day:'2-digit',hour:'2-digit',minute:'2-digit'}).formatToParts(i).map(x=>[x.type,x.value]));return {date:`${p.year}-${p.month}-${p.day}`,time:`${p.hour}:${p.minute}`}}
function nvm(date,lt,tz){let t=z2i(date,lt,tz);const z=i2z(t,tz);if(z.date===date&&z.time===lt)return t;for(let i=0;i<1440;i++){const prev=new Date(t.getTime()-60000);const pz=i2z(prev,tz);if(pz.date<date||(pz.date===date&&pz.time<lt))break;t=prev}return t}
