const settle = <T>(p: Promise<T>) =>
  p.then(
    (v) => ({ ok: true as const, v }),
    (e: unknown) => ({ ok: false as const, e: e as { code?: string; message?: string } }),
  );
type K = Awaited<ReturnType<typeof kandu>>;

const simAccount = async (org: ScenarioOrg, legalEntityId: string) =>
  org.run(org.setup, async (tx) => {
    const a = (await s.c.services.payments.tx.listAccounts(tx, legalEntityId)).find(
      (x) => x.provider === 'simulator',
    );
    if (!a) throw new Error('no simulator account');
    return a.id;
  });
const requestsOn = async (refId: string) =>
  (
    await sql<{ id: string; amount_minor: number; paid_minor: number; status: string; purpose: string }>`
      select id, amount_minor::int, paid_minor::int, status, purpose from payments.payment_requests
       where ref_id = ${refId} order by created_at`.execute(s.t.admin.kysely)
  ).rows;
const paymentsOn = async (refId: string) =>
  (
    await sql<{ amount_minor: number; status: string; method_id: string | null }>`
      select amount_minor::int, status, method_id from payments.payments where ref_id = ${refId}
       order by created_at`.execute(s.t.admin.kysely)
  ).rows;
const lateItems = async (folioId: string) =>
  (
    await sql<{ severity: string; status: string; message: string; details: Record<string, unknown> }>`
      select severity, status, message, details from finance.exceptions
       where kind = 'late_charge' and ref_id = ${folioId} order by created_at`.execute(s.t.admin.kysely)
  ).rows;

/** A checked-out stay (balance left open), its folio helpers and a late-charge poster (R2-D/E setup). */
async function departed(org: ScenarioOrg, k: K, token = 'pm_card_visa', inHouseChargeMinor = 0) {
  const desk = org.staff('front_desk', 'Hassan', [k.id]);
  const { reservations, finance } = s.c.services;
  const { profileId, card } = await guestWithCard(s, org, k, undefined, token);
  const b = applied<Booking>(
    await org.run(desk, (tx) =>
      reservations.createBooking(
        tx,
        bookingInput(
          k,
          { arrival: '2027-03-10', departure: '2027-03-12', unit: '214', guestProfileId: profileId },
          { guarantee: { kind: 'card', paymentMethodId: card.id } },
        ),
      ),
    ),
  );
  const stay = b.stays[0];
  if (!stay) throw new Error('no stay');
  await s.drain();
  await readyUnit(s, org, k.id, k.units['214'] as string);
  const checkIn = await reservations.performWorkflow(s.t.db, desk, {
    stayId: stay.id,
    kind: 'check_in',
    checkIn: { registration: { identityVerified: true }, access: { kind: 'card', count: 1 } },
  });
  expect(checkIn.run?.status).toBe('completed');
  await org.run(org.setup, (tx) => platform.advanceBusinessDate(tx, k.id, '2027-03-10'));
  if (inHouseChargeMinor > 0) {
    await org.run(desk, (tx) =>
      finance.postCharge(tx, {
        propertyId: k.id,
        target: { type: 'stay', id: stay.id },
        chargeCode: 'MISC',
        amount: { amountMinor: inHouseChargeMinor, currency: 'USD' },
        description: 'Boat transfer',
        serviceDate: '2027-03-11',
        sourceRef: { type: 'pos_check', id: newId() },
        idempotencyKey: `pos:${newId()}`,
      }),
    );
  }
  await org.run(org.setup, (tx) => platform.advanceBusinessDate(tx, k.id, '2027-03-11'));
  const manager = org.staff('general_manager', 'Duty manager', [k.id]);
  await reservations.performWorkflow(s.t.db, manager, {
    stayId: stay.id,
    kind: 'check_out',
    checkOut: { settle: { leaveOpenReason: 'Guest left early; settle by link' } },
  });
  await s.drain();
  const folioOf = async () =>
    (await org.run(desk, (tx) => finance.listFolios(tx, { owner: { type: 'stay', id: stay.id } })))[0];
  const late = (amountMinor: number, what: string) =>
    org.run(desk, (tx) =>
      finance.postCharge(tx, {
        propertyId: k.id,
        target: { type: 'stay', id: stay.id },
        chargeCode: 'MISC',
        amount: { amountMinor, currency: 'USD' },
        description: what,
        serviceDate: '2027-03-12',
        sourceRef: { type: 'pos_check', id: newId() },
        idempotencyKey: `pos:${newId()}`,
      }),
    );
  return { b, stay, desk, manager, card, profileId, folioOf, late };
}

describe('R3-B1 — invariant (b) writer not covered: settling the departed folio to receivables', () => {
  it('FINDING: the folio is direct-billed to the company, the guest link for the same debt stays open and can be paid', async () => {
    const org = await organization(s);
    const k = await kandu(s, org, '2027-03-10');
    const { finance, payments } = s.c.services;
    const { folioOf, late, manager, stay } = await departed(org, k);
    await late(20_000, 'Spa after departure');
    await s.drain();
    const folio = await folioOf();
    const owed = folio?.balance.amountMinor as number;
    const open0 = (await requestsOn(folio?.id as string)).filter((r) => r.status === 'open');
    expect(open0).toHaveLength(1);
    expect(open0[0]?.amount_minor).toBe(owed); // the R2-P1-2 fix holds up to here
    // Finance settles the departed folio to the company's receivables account (direct bill).
    const ar = await org.run(manager, (tx) =>
      finance.createReceivablesAccount(tx, {
        legalEntityId: k.legalEntityId,
        companyId: newId(),
        kind: 'company',
        name: 'Atoll Ventures Ltd',
        currency: 'USD',
        termsDays: 30,
        creditLimitMinor: 10_000_000,
      }),
    );
    const t = await org.run(manager, (tx) =>
      finance.transferToReceivables(tx, {
        folioId: folio?.id as string,
        receivablesAccountId: ar.id,
        idempotencyKey: `ar:${newId()}`,
      }),
    );
    await s.drain();
    expect(t.amount.amountMinor).toBe(owed);
    expect((await folioOf())?.balance.amountMinor).toBe(0);
    const open1 = (await requestsOn(folio?.id as string)).filter((r) => r.status === 'open');
    const items = await lateItems(folio?.id as string);
    console.log('R3-B1 after transfer to receivables:', JSON.stringify({ owed, open1, items }));
    // FINDING: the folio owes nothing (the company is invoiced), but the guest's link for the same money is open.
    expect(open1).toHaveLength(1);
    expect(open1[0]?.amount_minor).toBe(owed);
    expect(items.filter((i) => i.status === 'open')).toHaveLength(1); // urgent "collect" item still open
    // The guest pays the link: the same debt is now collected from the guest AND invoiced to the company.
    await payments.ensurePaymentLink(manager, open1[0]?.id as string);
    await payments.simulateRequestPayment(manager, open1[0]?.id as string);
    await payments.deliverSimulatorWebhooks(manager, await simAccount(org, k.legalEntityId));
    await s.drain();
    const paid = await paymentsOn(folio?.id as string);
    const after = await folioOf();
    const ledger = (
      await sql<{
        status: string;
        ledger_status: string | null;
      }>`select status, ledger_status from payments.payments
        where ref_id = ${folio?.id}`.execute(s.t.admin.kysely)
    ).rows;
    const allFolios = await org.run(manager, (tx) =>
      finance.listFolios(tx, { owner: { type: 'stay', id: stay.id } }),
    );
    const posted = (
      await sql<{
        folio_id: string;
        amount_minor: number;
        kind: string;
      }>`select p.folio_id, p.amount_minor::int, p.kind
        from finance.postings p where p.tenant_id = ${org.tenantId} and p.kind = 'payment'`.execute(
        s.t.admin.kysely,
      )
    ).rows;
    console.log(
      'R3-B1 ledger of the link payment:',
      JSON.stringify({
        ledger,
        allFolios: allFolios.map((f) => ({
          id: f.id,
          status: (f as { status?: unknown }).status,
          balance: f.balance,
        })),
        posted,
      }),
    );
    console.log(
      'R3-B1 after the guest paid the stale link:',
      JSON.stringify({ paid, balance: after?.balance }),
    );
    expect(paid.filter((p) => p.status === 'succeeded').map((p) => p.amount_minor)).toContain(owed);
    // FINDING: the guest's card paid the whole debt the company is also invoiced for (where it posts: logged above).
  });
});

describe('R3-B2 — invariant (b) exemption "defer while a payment is in flight" never re-syncs', () => {
  it('FINDING: a reversal during an unconfirmed payment leaves the link at the old amount and closes the item', async () => {
    const org = await organization(s);
    const k = await kandu(s, org, '2027-03-10');
    const { finance, payments } = s.c.services;
    const { folioOf, late, desk } = await departed(org, k);
    const posted = await late(20_000, 'Minibar (wrong room)');
    await s.drain();
    const folio = await folioOf();
    const owedBefore = folio?.balance.amountMinor as number;
    const [req] = (await requestsOn(folio?.id as string)).filter((r) => r.status === 'open');
    expect(req?.amount_minor).toBe(owedBefore);
    // The desk takes part of the room balance on a card whose provider answer never arrives (pending verification).
    const { card: slow } = await guestWithCard(
      s,
      org,
      k,
      { firstName: 'Paul', lastName: 'Pending' },
      'sim_timeout_then_failed',
    );
    const roomPart = owedBefore - 20_000;
    const r = await settle(
      payments.chargeNow(desk, {
        propertyId: k.id,
        methodId: slow.id,
        amount: { amountMinor: roomPart, currency: 'USD' },
        ref: { type: 'folio', id: folio?.id as string },
        separateCollection: { reason: 'Guest paid the room part on the phone' },
        businessKey: `pay:${newId()}`,
      }),
    );
    const inflight = await paymentsOn(folio?.id as string);
    console.log('R3-B2 desk payment:', r.ok ? r.v.status : r.e, JSON.stringify(inflight));
    expect(inflight.some((p) => p.status === 'pending_verification' || p.status === 'processing')).toBe(true);
    // The minibar charge was posted in error: the night auditor reverses it while the payment is unconfirmed.
    const auditor = org.staff('night_auditor', 'Auditor', [k.id]);
    await org.run(auditor, (tx) =>
      finance.reverseEntry(tx, {
        entryId: posted.entryId,
        reasonCode: 'posting_error',
        note: 'Wrong room',
        idempotencyKey: `rev:${newId()}`,
      }),
    );
    const afterReversal = {
      balance: (await folioOf())?.balance.amountMinor,
      requests: await requestsOn(folio?.id as string),
      items: await lateItems(folio?.id as string),
    };
    console.log('R3-B2 after reversal:', JSON.stringify(afterReversal));
    // FINDING (at once): the folio owes nothing, the late-charge item is closed, the link still asks for the charge.
    expect(afterReversal.balance).toBe(0);
    expect(afterReversal.requests.filter((x) => x.status === 'open').map((x) => x.amount_minor)).toEqual([
      owedBefore,
    ]);
    expect(afterReversal.items.filter((i) => i.status === 'open')).toEqual([]);
    await s.drain();
    // The payment is then confirmed by the provider ("check now"); nothing re-syncs the request.
    const pending = (
      await sql<{ id: string }>`select id from payments.payments where ref_id = ${folio?.id}
        and status in ('pending_verification', 'processing')`.execute(s.t.admin.kysely)
    ).rows;
    for (const p of pending) await settle(payments.checkNow(desk, { type: 'payment', id: p.id }));
    await s.drain();
    const final = {
      balance: (await folioOf())?.balance.amountMinor as number,
      payments: await paymentsOn(folio?.id as string),
      requests: (await requestsOn(folio?.id as string)).filter((x) => x.status === 'open'),
      items: (await lateItems(folio?.id as string)).filter((i) => i.status === 'open'),
    };
    console.log('R3-B2 after the payment was confirmed:', JSON.stringify(final));
    // FINDING: the link still asks for the pre-reversal amount, though the folio owes nothing (or is in credit).
    expect(final.balance).toBeLessThanOrEqual(0);
    expect(final.requests).toHaveLength(1);
    expect(final.requests[0]?.amount_minor).toBe(owedBefore);
  });
});

describe('R3-B3 — record the fix creates: a re-issued balance request leaves the old provider link payable', () => {
  it('FINDING: the guest pays the link they were sent first AND the re-issued one — the debt is collected twice', async () => {
    const org = await organization(s);
    const k = await kandu(s, org, '2027-03-10');
    const { payments } = s.c.services;
    const { folioOf, late, manager } = await departed(org, k);
    await late(20_000, 'Dinner');
    await s.drain();
    const folio = await folioOf();
    const [first] = (await requestsOn(folio?.id as string)).filter((r) => r.status === 'open');
    await payments.ensurePaymentLink(manager, first?.id as string); // the link the guest receives by email
    await late(10_000, 'Spa');
    await s.drain();
    const all = await requestsOn(folio?.id as string);
    const second = all.find((r) => r.status === 'open');
    expect(all.find((r) => r.id === first?.id)?.status).toBe('cancelled'); // re-issued (R2-P1-2 design)
    await payments.ensurePaymentLink(manager, second?.id as string);
    const owed = (await folioOf())?.balance.amountMinor as number;
    // The guest opens the FIRST email and pays it, then pays the second one too.
    const old = await settle(payments.simulateRequestPayment(manager, first?.id as string));
    await payments.simulateRequestPayment(manager, second?.id as string);
    await payments.deliverSimulatorWebhooks(manager, await simAccount(org, k.legalEntityId));
    await s.drain();
    const taken = (await paymentsOn(folio?.id as string)).filter((p) => p.status === 'succeeded');
    const balance = (await folioOf())?.balance.amountMinor as number;
    const alerts = (
      await sql<{ kind: string; severity: string }>`select kind, severity from payments.exceptions
        where tenant_id = ${org.tenantId} and kind = 'payment_request_overpaid'`.execute(s.t.admin.kysely)
    ).rows;
    console.log(
      'R3-B3:',
      JSON.stringify({ oldLinkPayable: old.ok ? 'yes' : old.e, owed, taken, balance, alerts }),
    );
    // FINDING: the cancelled request's provider link still takes money; the guest paid owed + the first amount.
    expect(old.ok).toBe(true);
    expect(taken.reduce((n, p) => n + p.amount_minor, 0)).toBeGreaterThan(owed);
  });
});

describe('R3-C1 — invariant (c) under concurrency: two legacy imports for the same booking id', () => {
  it('FINDING: two properties (two legal entities) import the same not-yet-migrated booking concurrently — both accepted into ONE account', async () => {
    const org = await organization(s);
    const k = await kandu(s, org, '2027-03-10');
    const h = await hiyani(s, org, '2027-03-10');
    const { finance } = s.c.services;
    const legacy = newId();
    const imp = (p: K) =>
      org.run(org.staff('finance', 'Finance', [p.id]), (tx) =>
        finance.importOpeningDeposits(tx, {
          propertyId: p.id,
          currency: 'USD',
          providedClosingMinor: 10_000,
          importRef: `legacy-${p.id.slice(-6)}`,
          items: [{ bookingId: legacy, amountMinor: 10_000, legacyNotMigrated: true }],
          idempotencyKey: `imp:${newId()}`,
        } as never),
      );
    const [a, b] = await Promise.allSettled([imp(k), imp(h)]);
    const accts = (
      await sql<{ property_id: string; balance_minor: number }>`select property_id, balance_minor::int
        from finance.deposit_accounts where booking_id = ${legacy}`.execute(s.t.admin.kysely)
    ).rows;
    const movements = (
      await sql<{ n: number }>`select count(*)::int as n from finance.deposit_movements
        where booking_id = ${legacy}`.execute(s.t.admin.kysely)
    ).rows[0]?.n;
    console.log(
      'R3-C1:',
      JSON.stringify({
        a: a.status === 'fulfilled' ? (a.value as { importedCount: number }).importedCount : String(a.reason),
        b: b.status === 'fulfilled' ? (b.value as { importedCount: number }).importedCount : String(b.reason),
        accts,
        movements,
      }),
    );
    // Sequentially the second import is rejected (builder's check); concurrently both pass the unlocked read, and the
    // second import's money is added to the FIRST property's account (another legal entity's deposit liability).
    expect(a.status === 'fulfilled' && b.status === 'fulfilled').toBe(true);
    expect(accts).toHaveLength(1);
    expect(accts[0]?.balance_minor).toBe(20_000); // FINDING: flip to "one import rejected, balance 10,000"
    expect(k.legalEntityId).not.toBe(h.legalEntityId);
  });
});
