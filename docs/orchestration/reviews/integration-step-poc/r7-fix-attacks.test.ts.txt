const settle = <T>(p: Promise<T>) =>
  p.then(
    (v) => ({ ok: true as const, v }),
    (e: unknown) => ({
      ok: false as const,
      e: e as { code?: string; message?: string; status?: number; details?: Record<string, unknown> },
    }),
  );
const codeOf = (r: { ok: boolean; e?: { code?: string } }) => (r.ok ? 'accepted' : (r.e?.code as string));
const sleep = (ms: number) => new Promise((r) => setTimeout(r, ms));
type K = Awaited<ReturnType<typeof kandu>>;
const usd = (amountMinor: number) => ({ amountMinor, currency: 'USD' });
function gate() {
  let open: () => void = () => undefined;
  const p = new Promise<void>((r) => {
    open = r;
  });
  return { p, open };
}

    return a.id;
  });

const cardCharge = (
  ctx: RequestContext,
  k: K,
  methodId: string,
  ref: { type: 'folio' | 'booking'; id: string },
  amountMinor: number,
  purpose?: 'balance' | 'deposit' | 'no_show' | 'custom' | 'incidentals',
) =>
  settle(
    s.c.services.payments.chargeNow(ctx, {
      propertyId: k.id,
      methodId,
      amount: usd(amountMinor),
      ref,
      ...(purpose ? { purpose } : {}),
      businessKey: newId(),
    }),
  );
async function addStay(
  org: ScenarioOrg,
  k: K,
  ctx: RequestContext,
  booking: Booking,
  guestProfileId: string,
) {
  const { reservations } = s.c.services;
  const extra = bookingInput(k, {
    arrival: booking.stays[0]?.arrival as string,
    departure: booking.stays[0]?.departure as string,
    type: 'WV',
    guestProfileId,
  }).stays[0];
  const preview = await org.run(ctx, (tx) =>
    reservations.previewChange(tx, { kind: 'add_stay', bookingId: booking.id, stay: extra } as never),
  );
  await reservations.commitChangeSafely(s.t.db, ctx, { previewToken: preview.token });
  const now = await org.run(ctx, (tx) => reservations.getBooking(tx, booking.id));
  return now.stays.find((x) => x.id !== booking.stays[0]?.id)?.id as string;
}

// ───────────────────────── R5-P1-1: purposes are facts ─────────────────────────
const penaltyPostingOf = async (stayId: string) =>
  (
    await sql<{ id: string; folio_id: string; amount_minor: number }>`
      select p.id, p.folio_id, p.amount_minor::int from finance.postings p
        join finance.journal_entries e on e.id = p.entry_id
       where e.idempotency_key in (${`penalty:no_show:${stayId}`}, ${`penalty:cancellation:${stayId}`})
         and p.kind = 'charge'`.execute(s.t.admin.kysely)
  ).rows[0];
const forfeitOf = (org: ScenarioOrg, ctx: RequestContext, bookingId: string, amountMinor: number) =>
  settle(
    org.run(ctx, (tx) =>
      s.c.services.finance.forfeitDeposit(tx, {
        bookingId,
        chargeCode: 'CXL',
        amountMinor,
        reason: 'Cancellation penalty',
        idempotencyKey: newId(),
      }),
    ),
  );
const heldOn = async (org: ScenarioOrg, bookingId: string) =>
  org.run(org.setup, (tx) => s.c.services.finance.depositHeldMinor(tx, bookingId));
async function cancelledWithCardDeposit(org: ScenarioOrg, k: K) {
  const { reservations } = s.c.services;
  const desk = org.staff('front_desk', 'Hassan', [k.id]);
  const gm = org.staff('general_manager', 'GM', [k.id]);
  const { profileId, card } = await guestWithCard(s, org, k);
  const b = await bookFor(org, k, desk, profileId, '2027-03-20', '2027-03-22');
  const stayId = b.stays[0]?.id as string;
  const folioId = await folioOfStay(org, k, stayId, profileId);
  await cardholderConfirms(s, org, card.id, b.id);
  expect(codeOf(await cardCharge(desk, k, card.id, { type: 'booking', id: b.id }, 10_000))).toBe('accepted');
  await s.drain();
  const preview = await org.run(gm, (tx) =>
    reservations.previewChange(tx, { kind: 'cancel_stay', stayId } as never),
  );
  await reservations.commitChangeSafely(s.t.db, gm, { previewToken: preview.token });
  await s.drain();
  const pen = await penaltyPostingOf(stayId);
  return { desk, gm, card, b, stayId, folioId, penalty: pen?.amount_minor ?? 0, penaltyPostingId: pen?.id };
}

// ───────────────────────── (d)/(b) forfeit rests on the record alone ─────────────────────────

describe('R7-F — a forfeit rests on the recorded penalty alone, not the penalty as it stands on the ledger', () => {
  it('FINDING R7-F1: deposit forfeited for the cancellation penalty, then the same penalty (still on the folio) charged to the card as no_show', async () => {
    const org = await organization(s);
    const k = await kandu(s, org, '2027-03-10');
    const x = await cancelledWithCardDeposit(org, k);
    expect(x.penalty).toBeGreaterThanOrEqual(10_000);
    expect(await balanceOf(org, x.folioId)).toBe(x.penalty); // the penalty is owed on the folio
    expect(codeOf(await forfeitOf(org, x.gm, x.b.id, 10_000))).toBe('accepted'); // deposit kept as revenue
    expect(await heldOn(org, x.b.id)).toBe(0);
    expect(await balanceOf(org, x.folioId)).toBe(x.penalty); // ...and the penalty is still owed in full
    const r = await cardCharge(x.desk, k, x.card.id, { type: 'folio', id: x.folioId }, x.penalty, 'no_show');
    expect(codeOf(r)).toBe('accepted'); // FINDING: the card has now paid penalty + 10,000 for one penalty
  });
  it('FINDING R7-F2 (reverse order): penalty charged to the card as no_show first, then the card deposit forfeited for it too', async () => {
    const org = await organization(s);
    const k = await kandu(s, org, '2027-03-10');
    const x = await cancelledWithCardDeposit(org, k);
    const owed = await balanceOf(org, x.folioId);
    const left = owed - (await heldOn(org, x.b.id)); // folioOwedNow nets deposits held
    expect(left).toBeGreaterThan(0);
    expect(
      codeOf(await cardCharge(x.desk, k, x.card.id, { type: 'folio', id: x.folioId }, left, 'no_show')),
    ).toBe('accepted');
    await s.drain();
    expect(codeOf(await forfeitOf(org, x.gm, x.b.id, 10_000))).toBe('accepted'); // FINDING
  });
  it('FINDING R7-F3: penalty credited off in full (goodwill), the card-funded deposit is still forfeited up to the record', async () => {
    const org = await organization(s);
    const k = await kandu(s, org, '2027-03-10');
    const x = await cancelledWithCardDeposit(org, k);
    const adj = await settle(
      org.run(x.gm, (tx) =>
        s.c.services.finance.adjustPosting(tx, {
          postingId: x.penaltyPostingId as string,
          reasonCode: 'goodwill',
          note: 'Penalty waived as goodwill',
          idempotencyKey: newId(),
        } as never),
      ),
    );
    expect(codeOf(adj)).toBe('accepted');
    expect(await balanceOf(org, x.folioId)).toBe(0);
    expect(codeOf(await forfeitOf(org, x.gm, x.b.id, 10_000))).toBe('accepted'); // FINDING
  });
});

    const auditor = org.staff('night_auditor', 'Night audit', [k.id]);
    const { profileId } = await guestWithCard(s, org, k);
    const b = await bookFor(org, k, desk, profileId, '2027-03-10', '2027-03-14');
    const stayId = b.stays[0]?.id as string;
    const folioId = await folioOfStay(org, k, stayId, profileId);
    const card = await methodFor(
      org,
      k,
      { type: 'guest_profile', id: profileId },

    await s.drain();
    const r = await cardCharge(desk, k, card.id, { type: 'folio', id: folioId }, penalty, 'no_show');
    expect(codeOf(r)).toBe('accepted'); // FINDING: the paid penalty still justifies taking an ordinary bill as no_show
  });
  it('CLEAN R7-N2: the penalty moved off the stay folio no longer justifies a no_show on it', async () => {
    const org = await organization(s);
    const k = await kandu(s, org, '2027-03-10');
    const { reservations, finance } = s.c.services;
    const desk = org.staff('front_desk', 'Hassan', [k.id]);
    const gm = org.staff('general_manager', 'GM', [k.id]);
    const auditor = org.staff('night_auditor', 'Night audit', [k.id]);

    const b = await bookFor(org, k, desk, profileId, '2027-03-10', '2027-03-14');
    const stayId = b.stays[0]?.id as string;
    const folioId = await folioOfStay(org, k, stayId, profileId);
    const card = await methodFor(
      org,
      k,
      { type: 'guest_profile', id: profileId },
      { purposes: ['guarantee', 'no_show'], bookingIds: [b.id] },
    );
    const noShow = await org.run(auditor, (tx) => reservations.markNoShow(tx, { stayId, idempotencyKey: newId() }));
    const penalty = noShow.penalty.amountMinor;
    await s.drain();
    const master = (
      await org.run(org.setup, (tx) =>
        financeForReservations(finance).openFolio(tx, {
          propertyId: k.id,
          owner: { type: 'booking', id: b.id },
          payer: { type: 'profile', id: profileId },
        }),
      )
    ).folioId;
    const t = await settle(
      org.run(gm, (tx) =>
        finance.transferPostings(tx, {
          fromFolioId: folioId,

    await s.drain();

describe('R7-C — capture: the label never steers the landing purpose; partial captures share one fact', () => {
  it('CLEAN R7-C1: a `deposit` label on a folio capture, and a `no_show` label on a booking capture, are refused; two partial no_show captures cannot exceed the penalty', async () => {
    const org = await organization(s);
    const k = await kandu(s, org, '2027-03-10');
    const { reservations, payments } = s.c.services;
    const desk = org.staff('front_desk', 'Hassan', [k.id]);
    const gm = org.staff('general_manager', 'GM', [k.id]);
    const auditor = org.staff('night_auditor', 'Night audit', [k.id]);
    const { profileId } = await guestWithCard(s, org, k);
    const b = await bookFor(org, k, desk, profileId, '2027-03-10', '2027-03-14');
    const stayId = b.stays[0]?.id as string;
    const folioId = await folioOfStay(org, k, stayId, profileId);
    const card = await methodFor(
      org,
      k,
      { type: 'guest_profile', id: profileId },
      { purposes: ['guarantee', 'no_show'], bookingIds: [b.id] },
    );
    const folio = { type: 'folio' as const, id: folioId };
    const hold = await org.run(desk, (tx) =>
      payments.tx.authorize(tx, {
        propertyId: k.id,
        methodId: card.id,
        amount: usd(400_000),
        ref: folio,
        businessKey: newId(),
      } as never),
    );
    await payments.deliverSimulatorWebhooks(gm, await simAccount(org, k.legalEntityId));
    await s.drain();
    const cap = (amountMinor: number, extra: Record<string, unknown>) =>
      settle(
        org.run(desk, (tx) =>
          payments.tx.capture(tx, {
            authorizationId: (hold as { id: string }).id,
            amount: usd(amountMinor),
            businessKey: newId(),
            ...extra,
          } as never),
        ),
      );
    await owe(org, k, desk, stayId, 50_000, 'Ordinary bill');
    await s.drain();
    expect(codeOf(await cap(50_000, { purpose: 'deposit' }))).toBe('mandate_purpose_not_covered');
    expect(codeOf(await cap(10_000, { purpose: 'no_show', ledgerRef: { type: 'booking', id: b.id } }))).toBe(
      'no_show_not_recorded',
    );
    const noShow = await org.run(auditor, (tx) => reservations.markNoShow(tx, { stayId, idempotencyKey: newId() }));
    const penalty = noShow.penalty.amountMinor;
    await s.drain();
    const half = Math.floor(penalty / 2) + 1;
    expect(codeOf(await cap(half, { purpose: 'no_show' }))).toBe('accepted');
    expect(codeOf(await cap(half, { purpose: 'no_show' }))).toBe('no_show_penalty_exceeded');
  });
});
