const settle = <T>(p: Promise<T>) =>
  p.then(
    (v) => ({ ok: true as const, v }),
    (e: unknown) => ({ ok: false as const, e }),
  );
const nightRole = (tenantId: string, environment: RequestContext['environment'], propertyId: string) => {
  const perms = [
    'platform.property.read',
    'reservations.booking.read',
    'reservations.stay.checkIn',
    'guests.profile.read',
    'guests.profile.update',
    'operations.unit.read',
    'inventory.unit.read',
  ];
  return {
    tenantId,
    environment,
    actor: { kind: 'user' as const, userId: newId(), displayName: 'Night receptionist' },
    authz: staticAuthorizer(perms.map((permission) => ({ permission, propertyIds: [propertyId] }))),
    correlationId: newId(),
  } satisfies RequestContext;
};

describe('R2-A — P1-2 reverse order: link the card owner to the booking first', () => {
  for (const variant of ['booker', 'sharing occupant'] as const) {
    it(`a reservations agent (no payments.payment.collect) makes the victim the ${variant}; a plain check-in holds the victim's card`, async () => {
      const org = await organization(s);
      const k = await kandu(s, org, '2027-03-10');
      const { reservations } = s.c.services;
      const desk = org.staff('front_desk', 'Hassan', [k.id]);
      // The victim: a returning guest whose card is on file from their own earlier booking.
      const { profileId: victim, card } = await guestWithCard(s, org, k, {
        firstName: 'Vera',
        lastName: 'Victim',
      });
      const own = applied<Booking>(
        await org.run(desk, (tx) =>
          reservations.createBooking(
            tx,
            bookingInput(
              k,
              {
                arrival: '2027-05-01',
                departure: '2027-05-03',
                guestProfileId: victim,
                guest: { firstName: 'Vera', lastName: 'Victim' },
              },
              { guarantee: { kind: 'card', paymentMethodId: card.id } },
            ),
          ),
        ),
      );
      const agent = org.staff('reservations_agent', 'Insider', [k.id]);
      expect(agent.authz.can('payments.payment.collect', { propertyId: k.id })).toBe(false);
      // Discovery: the agent reads the victim's booking and learns the profile id and the card's method id.
      const seen = await org.run(agent, (tx) => reservations.getBooking(tx, own.id));
      expect(seen.guarantee.paymentMethodId).toBe(card.id);
      expect(seen.booker.profileId).toBe(victim);
      // The agent books a long stay for an accomplice and links the victim.
      const base = bookingInput(k, {
        arrival: '2027-03-10',
        departure: '2027-03-17',
        unit: '214',
        guest: { firstName: 'Acc', lastName: 'Omplice' },
      });
      const stay0 = base.stays[0] as (typeof base.stays)[number];
      const input =
        variant === 'booker'
          ? {
              ...base,
              booker: {
                kind: 'profile' as const,
                profileId: victim,
                profile: { firstName: 'Vera', lastName: 'Victim' },
              },
              guarantee: { kind: 'card' as const, paymentMethodId: card.id },
            }
          : {
              ...base,
              guarantee: { kind: 'card' as const, paymentMethodId: card.id },
              stays: [
                {
                  ...stay0,
                  occupants: [
                    ...stay0.occupants,
                    {
                      profileId: victim,
                      profile: { firstName: 'Vera', lastName: 'Victim' },
                      kind: 'adult' as const,
                      primary: false,
                    },
                  ],
                },
              ],
            };
      const b = applied<Booking>(await org.run(agent, (tx) => reservations.createBooking(tx, input)));
      const stayId = b.stays[0]?.id as string;
      await s.drain();
      await readyUnit(s, org, k.id, k.units['214'] as string);
      // An ordinary check-in by a role with NO payments key, choosing nothing.
      const night = nightRole(org.tenantId, org.environment, k.id);
      const run = await reservations.performWorkflow(s.t.db, night, {
        stayId,
        kind: 'check_in',
        checkIn: { registration: { identityVerified: true }, access: { kind: 'card', count: 1 } },
      });
      const step = run.run?.steps.find((x) => x.key === 'guarantee');
      const gross = (await org.run(desk, (tx) => reservations.getBooking(tx, b.id))).stays[0]?.totals?.gross
        .amountMinor as number;
      const auths = await sql<{ method_id: string; amount_minor: number; status: string }>`
        select method_id, amount_minor, status from payments.authorizations where ref_id = ${stayId}`.execute(
        s.t.admin.kysely,
      );
      console.log(`R2-A ${variant}:`, JSON.stringify({ step, gross, auths: auths.rows }));
      // FINDING: the victim's card is held for the accomplice's 7-night stay, chosen by a role without payments keys.
      expect(step?.status).toBe('done');
      expect(auths.rows).toHaveLength(1);
      expect(auths.rows[0]).toMatchObject({ method_id: card.id, amount_minor: gross });
    });
  }
});

describe('R2-B — P1-2 guarantee cap: ±1 boundary and a non-numeric policy', () => {
  it('cap is allowed, cap + 1 is refused (nothing held)', async () => {
    const org = await organization(s);
    const k = await kandu(s, org, '2027-03-10');
    const { reservations } = s.c.services;
    const desk = org.staff('front_desk', 'Hassan', [k.id]);
    const { profileId, card } = await guestWithCard(s, org, k);
    const b = applied<Booking>(
      await org.run(desk, (tx) =>
        reservations.createBooking(
          tx,
          bookingInput(
            k,
            { arrival: '2027-03-10', departure: '2027-03-12', unit: '214', guestProfileId: profileId },
            { guarantee: { kind: 'card', paymentMethodId: card.id } },
          ),
        ),
      ),
    );
    const stayId = b.stays[0]?.id as string;
    const gross = b.stays[0]?.totals?.gross.amountMinor as number;
    const cap = Math.floor((gross * 150) / 100);
    await s.drain();
    await readyUnit(s, org, k.id, k.units['214'] as string);
    const auths = async () =>
      (
        await sql<{
          amount_minor: number;
        }>`select amount_minor from payments.authorizations where ref_id = ${stayId}`.execute(
          s.t.admin.kysely,
        )
      ).rows;
    const over = await reservations.performWorkflow(s.t.db, desk, {
      stayId,
      kind: 'check_in',
      checkIn: {
        registration: { identityVerified: true },
        guarantee: { amount: { amountMinor: cap + 1, currency: 'USD' } },
      },
    });
    expect(over.run?.steps.find((x) => x.key === 'guarantee')).toMatchObject({ status: 'failed' });
    expect(await auths()).toEqual([]);
    const at = await reservations.performWorkflow(s.t.db, desk, {
      stayId,
      kind: 'check_in',
      checkIn: {
        guarantee: { amount: { amountMinor: cap, currency: 'USD' } },
        access: { kind: 'card', count: 1 },
      },
    });
    expect(at.run?.steps.find((x) => x.key === 'guarantee')?.status).toBe('done');
    expect(await auths()).toEqual([{ amount_minor: cap }]);
  });

  it('a property policy value that is not a number removes the cap (fails open)', async () => {
    const org = await organization(s);
    const k = await kandu(s, org, '2027-03-10');

    const gm = org.staff('general_manager', 'GM', [k.id]);
    await org.run(gm, (tx) =>
      platform.setPolicy(tx, {
        key: 'reservations.checkIn.guaranteeCapPercent',
        scopeType: 'property',
        scopeId: k.id,
        value: 'no cap for VIPs',
        reason: 'typo in the settings screen',
      }),
    );
    const { profileId, card } = await guestWithCard(s, org, k);
    const b = applied<Booking>(
      await org.run(desk, (tx) =>
        reservations.createBooking(
          tx,
          bookingInput(
            k,
            { arrival: '2027-03-10', departure: '2027-03-11', unit: '214', guestProfileId: profileId },
            { guarantee: { kind: 'card', paymentMethodId: card.id } },
          ),
        ),
      ),
    );
    const stayId = b.stays[0]?.id as string;
    await s.drain();
    await readyUnit(s, org, k.id, k.units['214'] as string);
    const run = await reservations.performWorkflow(s.t.db, desk, {

      kind: 'check_in',

describe('R2-D — declared gap: several late charges on the same folio (concurrent and sequential)', () => {
  for (const sequential of [false, true])
    it(`${sequential ? 'sequential' : 'concurrent'}: three late charges on a folio in credit — requested total vs owed`, async () => {
      const org = await organization(s);
      const k = await kandu(s, org, '2027-03-10');
      const desk = org.staff('front_desk', 'Hassan', [k.id]);
      const { reservations, payments, finance } = s.c.services;
      const { profileId, card } = await guestWithCard(s, org, k);
      const b = applied<Booking>(
        await org.run(desk, (tx) =>
          reservations.createBooking(
            tx,
            bookingInput(
              k,
              { arrival: '2027-03-10', departure: '2027-03-12', unit: '214', guestProfileId: profileId },
              { guarantee: { kind: 'card', paymentMethodId: card.id } },
            ),
          ),
        ),
      );
      const stay = b.stays[0];
      if (!stay) throw new Error('no stay');
      await s.drain();
      await readyUnit(s, org, k.id, k.units['214'] as string);
      await payments.chargeNow(desk, {
        propertyId: k.id,
        methodId: card.id,
        amount: { amountMinor: 50_000, currency: 'USD' },
        ref: { type: 'booking', id: b.id },
        businessKey: `deposit:${b.id}:1`,
      });
      const checkIn = await reservations.performWorkflow(s.t.db, desk, {
        stayId: stay.id,
        kind: 'check_in',
        checkIn: { registration: { identityVerified: true }, access: { kind: 'card', count: 1 } },
      });
      expect(checkIn.run?.status).toBe('completed');
      await org.run(org.setup, (tx) => platform.advanceBusinessDate(tx, k.id, '2027-03-10'));
      await org.run(org.setup, (tx) => platform.advanceBusinessDate(tx, k.id, '2027-03-11'));
      const manager = org.staff('general_manager', 'Duty manager', [k.id]);
      await reservations.performWorkflow(s.t.db, manager, {
        stayId: stay.id,
        kind: 'check_out',
        checkOut: { settle: { leaveOpenReason: 'Deposit credit to be refunded' } },
      });
      await s.drain();
      const folioOf = async () =>
        (await org.run(desk, (tx) => finance.listFolios(tx, { owner: { type: 'stay', id: stay.id } })))[0];
      const folio = await folioOf();
      expect(folio?.balance.amountMinor).toBe(-50_000);
      const late = (amountMinor: number, what: string) =>
        org.run(desk, (tx) =>
          finance.postCharge(tx, {
            propertyId: k.id,
            target: { type: 'stay', id: stay.id },
            chargeCode: 'MISC',
            amount: { amountMinor, currency: 'USD' },
            description: what,
            serviceDate: '2027-03-12',
            sourceRef: { type: 'pos_check', id: newId() },
            idempotencyKey: `pos:${newId()}`,
          }),
        );
      // Each charge alone stays covered by the credit; together they exceed it.
      const results = sequential
        ? [
            { status: 'fulfilled' as const, value: await late(35_000, 'Spa A') },
            { status: 'fulfilled' as const, value: await late(35_000, 'Spa B') },
            { status: 'fulfilled' as const, value: await late(35_000, 'Spa C') },
          ]
        : await Promise.allSettled([late(35_000, 'Spa A'), late(35_000, 'Spa B'), late(35_000, 'Spa C')]);
      const exc = await sql<{ severity: string; message: string; details: unknown }>`
      select severity, message, details from finance.exceptions where kind = 'late_charge' and ref_id = ${folio?.id} order by created_at`.execute(
        s.t.admin.kysely,
      );
      console.log(
        'R2-D mode',
        sequential ? 'sequential' : 'concurrent',
        'exceptions:',
        JSON.stringify(exc.rows),
      );
      expect(results.every((r) => r.status === 'fulfilled')).toBe(true);
      const owed = (await folioOf())?.balance.amountMinor as number;
      const reqs = await sql<{ amount_minor: number }>`
      select amount_minor from payments.payment_requests where ref_type = 'folio' and ref_id = ${folio?.id}`.execute(
        s.t.admin.kysely,
      );
      const requested = reqs.rows.reduce((x, r) => x + Number(r.amount_minor), 0);
      console.log(
        `R2-D ${sequential ? 'sequential' : 'concurrent'} late charges:`,
        JSON.stringify({ owed, requests: reqs.rows }),
      );
      expect(owed).toBeGreaterThan(0);
      // FINDING: the 3rd charge's request is refused (an open balance link exists); the folio owes more than requested,
      // and the single open exception keeps the 'routine' severity of the first (covered-by-credit) charge.
      expect(requested).toBeLessThan(owed);
      expect(exc.rows).toEqual([expect.objectContaining({ severity: 'routine' })]);
    });
});

      ),

      finance.listFolios(tx, { owner: { type: 'stay', id: stay.id } }),
    );
    const before = folio?.balance.amountMinor as number;
    console.log('R2-E checkout:', out.run?.status, 'balance', before);
    const posted = await org.run(desk, (tx) =>
      finance.postCharge(tx, {
        propertyId: k.id,
        target: { type: 'stay', id: stay.id },
        chargeCode: 'MISC',
        amount: { amountMinor: 20_000, currency: 'USD' },
        description: 'Minibar (wrong room)',
        serviceDate: '2027-03-11',
        sourceRef: { type: 'pos_check', id: newId() },
        idempotencyKey: `pos:${newId()}`,
      }),
    );
    const auditor = org.staff('night_auditor', 'Auditor', [k.id]);

      finance.reverseEntry(tx, {
        entryId: posted.entryId,
        reasonCode: 'posting_error',
        note: 'Charged to the wrong room',
        idempotencyKey: `rev:${newId()}`,
      }),

      finance.listFolios(tx, { owner: { type: 'stay', id: stay.id } }),

      s.t.admin.kysely,
    );
    console.log(
      'R2-E after reversal:',
      JSON.stringify({ before, after: after?.balance, requests: reqs.rows }),
    );
    expect(after?.balance.amountMinor).toBe(before);
    // FINDING: the request for the reversed charge is still open.
    expect(reqs.rows.filter((r) => r.status === 'open').length).toBeGreaterThan(0);
  });
});
