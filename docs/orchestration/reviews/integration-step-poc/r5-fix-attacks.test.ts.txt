let s: Scenario;

  s = await startScenario();

afterAll(async () => s?.close());

const settle = <T>(p: Promise<T>) =>
  p.then(
    (v) => ({ ok: true as const, v }),
    (e: unknown) => ({ ok: false as const, e: e as { code?: string; message?: string } }),
  );
const codeOf = (r: { ok: boolean; e?: { code?: string } }) => (r.ok ? 'accepted' : (r.e?.code as string));
const sleep = (ms: number) => new Promise((r) => setTimeout(r, ms));
type K = Awaited<ReturnType<typeof kandu>>;
const usd = (amountMinor: number) => ({ amountMinor, currency: 'USD' });

const simAccount = async (org: ScenarioOrg, legalEntityId: string) =>
  org.run(org.setup, async (tx) => {
    const a = (await s.c.services.payments.tx.listAccounts(tx, legalEntityId)).find(
      (x) => x.provider === 'simulator',
    );
    if (!a) throw new Error('no simulator account');
    return a.id;
  });
const requestsOn = async (refId: string) =>
  (
    await sql<{ id: string; amount_minor: number; paid_minor: number; status: string; purpose: string }>`
      select id, amount_minor::int, paid_minor::int, status, purpose from payments.payment_requests
       where ref_id = ${refId} order by created_at`.execute(s.t.admin.kysely)
  ).rows;
const paymentsOn = async (refId: string) =>
  (
    await sql<{ amount_minor: number; status: string; method_id: string | null }>`
      select amount_minor::int, status, method_id from payments.payments where ref_id = ${refId}
       order by created_at`.execute(s.t.admin.kysely)
  ).rows;
const lateItems = async (folioId: string) =>
  (
    await sql<{ severity: string; status: string; message: string }>`
      select severity, status, message from finance.exceptions
       where kind = 'late_charge' and ref_id = ${folioId} order by created_at`.execute(s.t.admin.kysely)
  ).rows;
const linkStates = async (refId: string) =>
  (
    await sql<{
      id: string;
      status: string;
      link_status: string;
      amount_minor: number;
      link_url: string | null;
    }>`
      select id, status, link_status, amount_minor::int, link_url from payments.payment_requests
       where ref_id = ${refId} order by created_at`.execute(s.t.admin.kysely)
  ).rows;
const payExceptions = async (org: ScenarioOrg) =>
  (
    await sql<{
      kind: string;
      severity: string;
      status: string;
      owner_role: string | null;
      next_step: string;
      business_impact: string;
      ref_id: string;
    }>`select kind, severity, status, owner_role, next_step, business_impact, ref_id from payments.exceptions
       where tenant_id = ${org.tenantId} order by created_at`.execute(s.t.admin.kysely)
  ).rows;
const methodFor = async (
  org: ScenarioOrg,
  k: K,
  owner: { type: 'guest_profile' | 'booking'; id: string },
  scope: Record<string, unknown>,
  token = 'pm_card_amex',
) =>
  org.run(org.setup, async (tx) =>
    s.c.services.payments.tx.createMethod(tx, {
      accountId: await simAccount(org, k.legalEntityId),
      propertyId: k.id,
      kind: 'card',
      providerToken: token,
      brand: 'amex',
      last4: '8431',
      owner,
      mitAgreement: true,
      mandate: {
        termsVersion: 'card-on-file-2027-01',
        acceptedAt: s.clock.now().toISOString(),
        channel: 'front_desk',
        scope: { legalEntityId: k.legalEntityId, ...scope } as never,
      },
    }),
  );
const bookFor = async (
  org: ScenarioOrg,
  k: K,
  ctx: RequestContext,
  profileId: string,
  arrival = '2027-03-20',
  departure = '2027-03-22',
  type = 'BV',
) =>
  applied<Booking>(
    await org.run(ctx, (tx) =>
      s.c.services.reservations.createBooking(
        tx,
        bookingInput(k, { arrival, departure, guestProfileId: profileId, type }),
      ),
    ),
  );
const folioOfStay = async (org: ScenarioOrg, k: K, stayId: string, payer: string) =>
  (
    await org.run(org.setup, (tx) =>
      financeForReservations(s.c.services.finance).openFolio(tx, {
        propertyId: k.id,
        owner: { type: 'stay', id: stayId },
        payer: { type: 'profile', id: payer },
      }),
    )
  ).folioId;
const holdOn = async (
  org: ScenarioOrg,
  k: K,
  ctx: RequestContext,
  methodId: string,
  ref: object,
  amountMinor: number,
) => {
  const { payments } = s.c.services;
  const r = await settle(
    org.run(ctx, (tx) =>
      payments.tx.authorize(tx, {
        propertyId: k.id,
        methodId,
        amount: usd(amountMinor),
        ref: ref as { type: 'stay'; id: string },
        businessKey: newId(),
      }),
    ),
  );
  if (r.ok) await payments.settle(ctx, { type: 'authorization', id: r.v.id });
  return r;
};

/** A checked-out stay (balance left open), its folio helpers and a late-charge poster (R2-D/E setup). */
async function departed(org: ScenarioOrg, k: K) {
  const desk = org.staff('front_desk', 'Hassan', [k.id]);
  const { reservations, finance } = s.c.services;
  const { profileId, card } = await guestWithCard(s, org, k);
  const b = await bookWithCard(
    s,
    org,
    desk,
    bookingInput(
      k,
      { arrival: '2027-03-10', departure: '2027-03-12', unit: '214', guestProfileId: profileId },
      { guarantee: { kind: 'card', paymentMethodId: card.id } },
    ),
  );
  const stay = b.stays[0];
  if (!stay) throw new Error('no stay');
  await s.drain();
  await readyUnit(s, org, k.id, k.units['214'] as string);
  const checkIn = await reservations.performWorkflow(s.t.db, desk, {
    stayId: stay.id,
    kind: 'check_in',
    checkIn: { registration: { identityVerified: true }, access: { kind: 'card', count: 1 } },
  });
  expect(checkIn.run?.status).toBe('completed');
  await org.run(org.setup, (tx) => platform.advanceBusinessDate(tx, k.id, '2027-03-10'));
  await org.run(org.setup, (tx) => platform.advanceBusinessDate(tx, k.id, '2027-03-11'));
  const manager = org.staff('general_manager', 'Duty manager', [k.id]);
  await reservations.performWorkflow(s.t.db, manager, {
    stayId: stay.id,
    kind: 'check_out',
    checkOut: { settle: { leaveOpenReason: 'Guest left early; settle by link' } },
  });
  await s.drain();
  const folioOf = async () =>
    (await org.run(desk, (tx) => finance.listFolios(tx, { owner: { type: 'stay', id: stay.id } })))[0];
  const late = (amountMinor: number, what: string) =>
    org.run(desk, (tx) =>
      finance.postCharge(tx, {
        propertyId: k.id,
        target: { type: 'stay', id: stay.id },
        chargeCode: 'MISC',
        amount: usd(amountMinor),
        description: what,
        serviceDate: '2027-03-12',
        sourceRef: { type: 'pos_check', id: newId() },
        idempotencyKey: `pos:${newId()}`,
      }),
    );
  return { b, stay, desk, manager, card, profileId, folioOf, late };
}

// ───────────────────────── R4-P1-1: the agreement covers only the bookings it names ─────────────────────────

async function namedBookingWithFolio(org: ScenarioOrg, k: K, arrival = '2027-07-01', departure = '2027-07-03') {
  const desk = org.staff('front_desk', 'Hassan', [k.id]);
  const { profileId, card } = await guestWithCard(s, org, k);
  const b = await bookFor(org, k, desk, profileId, arrival, departure);
  await cardholderConfirms(s, org, card.id, b.id);
  const folioId = await folioOfStay(org, k, b.stays[0]?.id as string, profileId);
  return { desk, profileId, card, b, folioId };
}

describe('R5-A — `no_show` is a label the charging role chooses', () => {
  it('FINDING R5-A1: a guarantee + no-show agreement lets the desk take USD 2,500 labelled no_show from a booking that is not a no-show', async () => {
    const org = await organization(s);
    const k = await kandu(s, org, '2027-03-10');
    const { payments } = s.c.services;
    const desk = org.staff('front_desk', 'Hassan', [k.id]);
    const { profileId } = await guestWithCard(s, org, k);
    const b = await bookFor(org, k, desk, profileId, '2027-03-10', '2027-03-14');
    const folioId = await folioOfStay(org, k, b.stays[0]?.id as string, profileId);
    const card = await methodFor(
      org,
      k,
      { type: 'guest_profile', id: profileId },
      { purposes: ['guarantee', 'no_show'], bookingIds: [b.id] },
    );
    const take = (purpose: 'balance' | 'no_show') =>
      settle(
        payments.chargeNow(desk, {
          propertyId: k.id,
          methodId: card.id,
          amount: usd(250_000),
          ref: { type: 'folio', id: folioId },
          purpose,
          businessKey: newId(),
        }),
      );
    // The cardholder agreed to a guarantee and a no-show fee, not to the balance: refused as balance ...
    expect(codeOf(await take('balance'))).toBe('mandate_purpose_not_covered');
    const stay = (
      await sql<{ status: string }>`select status from reservations.stays where id = ${b.stays[0]?.id as string}`.execute(
        s.t.admin.kysely,
      )
    ).rows[0];
    expect(stay?.status).not.toBe('no_show');
    // ... and taken in full when the desk calls it a no-show fee.
    const r = await take('no_show');
    expect(codeOf(r)).toBe('accepted'); // FINDING
    expect((await paymentsOn(folioId)).filter((p) => p.status === 'succeeded').map((p) => p.amount_minor)).toEqual([
      250_000,
    ]);
  });

  it('FINDING R5-A2: the R4-J double collection reopens with the `no_show` label (paid deposit link, deposit not applied, no reason)', async () => {
    const org = await organization(s);
    const k = await kandu(s, org, '2027-03-10');
    const { payments } = s.c.services;
    const gm = org.staff('general_manager', 'GM', [k.id]);
    const { desk, profileId, card, b, folioId } = await namedBookingWithFolio(org, k);
    const req = await org.run(desk, (tx) =>
      payments.tx.createPaymentRequest(tx, {
        propertyId: k.id,
        payer: { type: 'guest_profile', id: profileId, name: 'Elena Ivanova' },
        amount: usd(30_000),
        purpose: 'deposit',
        ref: { type: 'booking', id: b.id },
        expiresAt: new Date(s.clock.now().getTime() + 86_400_000).toISOString(),
        businessKey: newId(),
      }),
    );
    await payments.ensurePaymentLink(gm, req.id);
    await payments.simulateRequestPayment(gm, req.id);
    await payments.deliverSimulatorWebhooks(gm, await simAccount(org, k.legalEntityId));
    const take = (purpose: 'balance' | 'no_show') =>
      settle(
        payments.chargeNow(desk, {
          propertyId: k.id,
          methodId: card.id,
          amount: usd(30_000),
          ref: { type: 'folio', id: folioId },
          purpose,
          businessKey: newId(),
        }),
      );
    expect(codeOf(await take('balance'))).toBe('collection_owned_by_payment_request'); // R4-J fix holds
    expect(codeOf(await take('no_show'))).toBe('accepted'); // FINDING: same money, new label, no reason
  });
});

describe('R5-B — the running total at N=10, mixed uses', () => {
  for (const round of [1, 2, 3]) {
    it(`CLEAN R5-B round ${round}: capture of a 60 hold + 4 charges + 5 holds of 40 at once on a 100 agreement stay ≤ 100`, async () => {
      const org = await organization(s);
      const k = await kandu(s, org, '2027-03-10');
      const { payments } = s.c.services;
      const desk = org.staff('front_desk', 'Hassan', [k.id]);
      const { profileId } = await guestWithCard(s, org, k);
      const bookings = [];
      for (let i = 0; i < 10; i++)
        bookings.push(
          await bookFor(org, k, desk, profileId, `2027-05-${String(10 + i * 2)}`, `2027-05-${String(11 + i * 2)}`),
        );
      const card = await methodFor(
        org,
        k,
        { type: 'guest_profile', id: profileId },
        { purposes: ['guarantee', 'deposit'], maxAmount: usd(10_000), bookingIds: bookings.map((b) => b.id) },
      );
      const b0 = bookings[0] as Booking;
      const hold = await holdOn(org, k, desk, card.id, { type: 'booking', id: b0.id }, 6_000);
      expect(hold.ok).toBe(true);
      const holdId = (hold as { v: { id: string } }).v.id;
      const ops: Array<Promise<{ ok: boolean; e?: { code?: string } }>> = [
        settle(
          org.run(desk, (tx) =>
            payments.tx.capture(tx, {
              authorizationId: holdId,
              ledgerRef: { type: 'booking', id: b0.id },
              businessKey: newId(),
            }),
          ),
        ),
      ];
      for (let i = 1; i < 5; i++)
        ops.push(
          settle(
            payments.chargeNow(desk, {
              propertyId: k.id,
              methodId: card.id,
              amount: usd(4_000),
              ref: { type: 'booking', id: (bookings[i] as Booking).id },
              purpose: 'deposit',
              businessKey: newId(),
            }),
          ),
        );
      for (let i = 5; i < 10; i++)
        ops.push(holdOn(org, k, desk, card.id, { type: 'booking', id: (bookings[i] as Booking).id }, 4_000));
      const rs = await Promise.all(ops);
      await s.drain();
      const taken = Number(
        (
          await sql<{ n: string }>`select coalesce(sum(amount_minor), 0) as n from payments.payments
          where method_id = ${card.id} and initiator = 'merchant' and status in ('processing','pending_verification','succeeded')`.execute(
            s.t.admin.kysely,
          )
        ).rows[0]?.n,
      );
      const open = Number(
        (
          await sql<{ n: string }>`select coalesce(sum(amount_minor - captured_minor - capture_reserved_minor), 0) as n
          from payments.authorizations where method_id = ${card.id}
          and status in ('processing','pending_verification','authorized','partially_captured')`.execute(s.t.admin.kysely)
        ).rows[0]?.n,
      );
      const codes = rs.map(codeOf);
      expect(taken + open, JSON.stringify(codes)).toBeLessThanOrEqual(10_000);
      expect(codes.filter((c) => c === 'deadlock_detected' || /deadlock/i.test(c ?? ''))).toEqual([]);
    });
  }
});

describe('R5-C — staff "link stopped" on a link the provider reported as already paid', () => {
  it('FINDING R5-C: the desk confirms a `completed` link stopped — the replacement asks for money the guest has just paid, and the guest pays twice', async () => {
    const org = await organization(s);
    const k = await kandu(s, org, '2027-03-10');
    const { payments } = s.c.services;
    const { folioOf, late, manager, desk } = await departed(org, k);
    await late(20_000, 'Dinner');
    await s.drain();
    const folio = await folioOf();
    const [first] = (await requestsOn(folio?.id as string)).filter((r) => r.status === 'open');
    await payments.ensurePaymentLink(manager, first?.id as string);
    await payments.simulateRequestPayment(manager, first?.id as string); // paid at the provider, webhook not yet here
    await late(10_000, 'Spa');
    await s.drain();
    const waiting = await linkStates(folio?.id as string);
    expect(waiting.find((r) => r.id === first?.id)?.link_status).toBe('completed');
    expect(waiting.find((r) => r.status === 'open')).toMatchObject({ link_status: 'pending', link_url: null });
    // No exception exists yet (deadline not reached); the front desk confirms "stopped" anyway.
    expect((await payExceptions(org)).filter((e) => e.kind === 'payment_link_not_stopped')).toEqual([]);
    const r = await settle(
      org.run(desk, (tx) =>
        payments.tx.confirmLinkStopped(tx, { requestId: first?.id as string, reason: 'Stopped it' }),
      ),
    );
    expect(codeOf(r)).toBe('accepted'); // FINDING (front desk; provider said it was PAID, not stopped)
    await s.drain();
    const now = await linkStates(folio?.id as string);
    const replacement = now.find((r) => r.status === 'open');
    expect(now.find((r) => r.id === first?.id)?.link_status).toBe('deactivated');
    expect(replacement?.link_status).toBe('created'); // FINDING: released before the payment is recorded
    const owedAtProvider = (folio?.balance.amountMinor as number) + 10_000 - (first?.amount_minor as number);
    expect(replacement?.amount_minor).toBeGreaterThan(owedAtProvider); // asks again for the 20,000 just paid
    // The guest pays the link they were sent; then the first payment's webhook arrives.
    await payments.simulateRequestPayment(manager, replacement?.id as string);
    await payments.deliverSimulatorWebhooks(manager, await simAccount(org, k.legalEntityId));
    await s.drain();
    const after = (await folioOf())?.balance.amountMinor as number;
    expect(after).toBeLessThan(0); // FINDING: the guest is in credit — paid twice
    const overpaid = (await payExceptions(org)).filter((e) => e.kind === 'payment_request_overpaid');
    expect(overpaid.length).toBeGreaterThan(0);
  });
});

describe('R5-D — a stay added after consent, paid through the booking-level reference', () => {
  it('FINDING R5-D: after add_stay (Otto) on Vera’s named booking, the desk charges Vera’s card a booking “deposit” and applies it to Otto’s stay folio', async () => {
    const org = await organization(s);
    const k = await kandu(s, org, '2027-03-10');
    const { reservations, payments, finance } = s.c.services;
    const desk = org.staff('front_desk', 'Hassan', [k.id]);
    const { profileId: vera, card } = await guestWithCard(s, org, k, { firstName: 'Vera', lastName: 'Victim' });
    const { profileId: otto } = await guestWithCard(s, org, k, { firstName: 'Otto', lastName: 'Other' });
    const v = await bookFor(org, k, desk, vera, '2027-06-10', '2027-06-12');
    await cardholderConfirms(s, org, card.id, v.id);
    const extra = bookingInput(k, {
      arrival: '2027-06-10',
      departure: '2027-06-12',
      type: 'WV',
      guestProfileId: otto,
    }).stays[0];
    const preview = await org.run(desk, (tx) =>
      reservations.previewChange(tx, { kind: 'add_stay', bookingId: v.id, stay: extra } as never),
    );
    await reservations.commitChangeSafely(s.t.db, desk, { previewToken: preview.token });
    const now = await org.run(desk, (tx) => reservations.getBooking(tx, v.id));
    const addedId = now.stays.find((x) => x.id !== v.stays[0]?.id)?.id as string;
    const ottoFolio = await folioOfStay(org, k, addedId, otto);
    // The added stay itself is refused (the builder's fix) ...
    const direct = await settle(
      payments.chargeNow(desk, {
        propertyId: k.id,
        methodId: card.id,
        amount: usd(250_000),
        ref: { type: 'folio', id: ottoFolio },
        businessKey: newId(),
      }),
    );
    expect(codeOf(direct)).toBe('mandate_reference_not_covered');
    // ... but the booking-level reference has no stay, so it is covered whatever stays the booking has now.
    const viaBooking = await settle(
      payments.chargeNow(desk, {
        propertyId: k.id,
        methodId: card.id,
        amount: usd(250_000),
        ref: { type: 'booking', id: v.id },
        purpose: 'deposit',
        businessKey: newId(),
      }),
    );
    expect(codeOf(viaBooking)).toBe('accepted'); // FINDING
    await s.drain();
    const moved = await settle(
      org.run(desk, (tx) =>
        finance.applyDeposits(tx, { bookingId: v.id, toFolioId: ottoFolio, idempotencyKey: newId() }),
      ),
    );
    console.log('R5-D applyDeposits to Otto’s added-stay folio:', codeOf(moved), moved.ok ? moved.v.applied : '');
  });
});
